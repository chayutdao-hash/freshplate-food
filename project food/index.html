⚡ ตั้งค่า Supabase (PostgreSQL & Authentication)

ตรวจสอบสถานะ...
ใช้ Supabase Cloud PostgreSQL บันทึกข้อมูลเมนูอาหาร & Authentication
📖 วิธีเชื่อมต่อ Supabase ใน 3 นาที: เปิด Supabase Dashboard ↗
  1. สร้างโปรเจกต์ใหม่ใน Supabase Dashboard (ฟรี)
  2. ไปที่ Project Settings (ฟันเฟือง) > API > คัดลอก Project URL และ anon public Key
  3. วาง URL และ Key ด้านล่างนี้ หรือวางโค้ด createClient(...)
  4. ไปที่ SQL Editor ใน Supabase > รัน SQL Script ด้านล่างเพื่อสร้างตารางพร้อมใช้งานทันที!
✏️ กรอกแยกทีละช่อง (Project URL & Anon Key)
📋 ดูโค้ดสร้างตาราง (SQL Script สำหรับ Supabase SQL Editor) 
-- 1. ตารางโปรไฟล์ผู้ใช้งาน
CREATE TABLE IF NOT EXISTS public.users (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT,
  photo_url TEXT,
  role TEXT NOT NULL DEFAULT 'user',
  target_calories INTEGER NOT NULL DEFAULT 2000,
  target_protein INTEGER NOT NULL DEFAULT 130,
  target_carbs INTEGER NOT NULL DEFAULT 210,
  target_fat INTEGER NOT NULL DEFAULT 65,
  health_goal TEXT DEFAULT 'รักษาสุขภาพทั่วไป',
  created_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW()),
  updated_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW())
);

-- 2. ตารางแผนอาหารและสูตรอาหารที่บันทึก
CREATE TABLE IF NOT EXISTS public.user_data (
  user_id TEXT PRIMARY KEY,
  meal_plan JSONB,
  saved_recipes JSONB,
  updated_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW())
);

-- 3. ตารางคำปรึกษาโภชนาการ
CREATE TABLE IF NOT EXISTS public.consultations (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id TEXT NOT NULL,
  user_name TEXT,
  user_email TEXT,
  user_photo TEXT,
  user_target_cal INTEGER DEFAULT 2000,
  subject TEXT NOT NULL,
  health_goal TEXT,
  question TEXT NOT NULL,
  admin_reply TEXT,
  replied_by_admin_id TEXT,
  admin_name TEXT,
  status TEXT NOT NULL DEFAULT 'pending',
  created_at TIMESTAMPTZ DEFAULT TIMEZONE('utc'::text, NOW()),
  replied_at TIMESTAMPTZ
);

-- เปิด Row Level Security (RLS) และเปิดสิทธิ์การใช้งาน
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.user_data ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.consultations ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow public read-write for users" ON public.users FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public read-write for user_data" ON public.user_data FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Allow public read-write for consultations" ON public.consultations FOR ALL USING (true) WITH CHECK (true);
